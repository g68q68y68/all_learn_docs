# OpenCV+Tesseract OCR数字识别优化方案：基于轮廓分割与修复

针对小图像数字断续导致的识别问题，我将提供一个完整的解决方案，结合轮廓分割、图像修复和形态学操作来提高识别准确率。

```java
    private static List<DelayResult> processVideo(String videoPath, Rect textRegion, String tessdataPath) {
        List<DelayResult> results = new ArrayList<>();
      
        try {
            // 初始化OCR引擎
            ITesseract tesseract = initializeOCR(tessdataPath);
          
            // 打开视频
            VideoCapture capture = new VideoCapture(videoPath);
            if (!capture.isOpened()) {
                System.err.println("无法打开视频文件: " + videoPath);
                return results;
            }
          
            // 获取视频属性
            double fps = Math.max(capture.get(VideoCapture.CAP_PROP_FPS), 30);
            int totalFrames = (int) capture.get(VideoCapture.CAP_PROP_FRAME_COUNT);
            int frameSkip = (int) Math.round(fps); // 每秒取一帧
          
            Mat frame = new Mat();
            int frameIndex = 0;
          
            while (true) {
                // 读取帧
                if (!capture.read(frame) || frame.empty()) break;
              
                // 每秒处理一次
                if (frameIndex % frameSkip == 0) {
                    // 计算当前时间（秒）
                    double currentTime = frameIndex / fps;
                  
                    // 截取区域
                    Mat roi = new Mat(frame, textRegion);
                  
                    // 图像增强
                    Mat enhanced = enhanceTextRegion(roi);
                  
                    // 修复和分割数字
                    Mat prepared = prepareDigits(enhanced);
                  
                    // OCR识别 - 尝试两种方法
                    String recognizedText = recognizeWithCombinedStrategy(prepared, tesseract);
                  
                    // 提取数字部分
                    String delayValue = extractDelayValue(recognizedText);
                  
                    // 添加到结果
                    results.add(new DelayResult(currentTime, delayValue));
                  
                    // 输出进度
                    System.out.printf("时间: %.1fs | 延迟: %s\n", currentTime, delayValue);
                  
                    // 调试：保存处理图像
                    if (frameIndex % (5 * frameSkip) == 0) {
                        saveDebugImage(roi, "raw_" + frameIndex, false);
                        saveDebugImage(enhanced, "enhanced_" + frameIndex, true);
                        saveDebugImage(prepared, "segmented_" + frameIndex, true);
                    }
                }
              
                frameIndex++;
            }
            capture.release();
        } catch (Exception e) {
            e.printStackTrace();
        }
      
        return results;
    }

    // 图像增强
    private static Mat enhanceTextRegion(Mat region) {
        // 1. 放大图像 (8倍)
        double scaleFactor = 8.0;
        Mat scaled = new Mat();
        Imgproc.resize(region, scaled, new Size(), scaleFactor, scaleFactor, Imgproc.INTER_CUBIC);
      
        // 2. 转换为灰度
        Mat gray = new Mat();
        Imgproc.cvtColor(scaled, gray, Imgproc.COLOR_BGR2GRAY);
      
        // 3. 对比度增强：非局部均值去噪
        Mat denoised = new Mat();
        Imgproc.fastNlMeansDenoising(gray, denoised, 10, 7, 21);
      
        // 4. 自适应直方图均衡化
        Mat equalized = new Mat();
        Imgproc.createCLAHE(3.0, new Size(10, 10)).apply(denoised, equalized);
      
        return equalized;
    }
  
    // 数字分割和修复
    private static Mat prepareDigits(Mat processed) {
        // 1. 二值化
        Mat binary = new Mat();
        Imgproc.adaptiveThreshold(
            processed, binary, 255,
            Imgproc.ADAPTIVE_THRESH_GAUSSIAN_C, 
            Imgproc.THRESH_BINARY_INV, 
            21, 
            5
        );
      
        // 2. 形态学操作 - 连接数字断裂
        Mat kernel = Imgproc.getStructuringElement(Imgproc.MORPH_ELLIPSE, new Size(3, 3));
        Imgproc.morphologyEx(binary, binary, Imgproc.MORPH_CLOSE, kernel);
      
        // 3. 查找轮廓
        List<MatOfPoint> contours = new ArrayList<>();
        Mat hierarchy = new Mat();
        Imgproc.findContours(binary, contours, hierarchy, Imgproc.RETR_EXTERNAL, Imgproc.CHAIN_APPROX_SIMPLE);
      
        // 4. 创建纯黑背景掩模
        Mat mask = Mat.zeros(binary.size(), CvType.CV_8UC1);
      
        // 5. 筛选并填充数字轮廓
        for (MatOfPoint contour : contours) {
            MatOfPoint2f curve = new MatOfPoint2f(contour.toArray());
          
            // 轮廓面积过滤
            double area = Imgproc.contourArea(curve);
            if (area < 100 || area > 5000) continue;
          
            // 填充轮廓以修复断裂
            Imgproc.drawContours(mask, Collections.singletonList(contour), -1, new Scalar(255), Imgproc.FILLED);
        }
      
        // 6. 图像修复 - 使用inpainting填补小孔洞
        Mat inpainted = new Mat();
        Photo.inpaint(mask, mask, inpainted, 3, Photo.INPAINT_TELEA);
      
        // 7. 形态学平滑
        Imgproc.morphologyEx(inpainted, inpainted, Imgproc.MORPH_OPEN, kernel);
      
        return inpainted;
    }
  
    // 双重识别策略
    private static String recognizeWithCombinedStrategy(Mat prepared, ITesseract tesseract) {
        // 第一种策略: 整体识别
        String result1 = performOCR(prepared, tesseract);
      
        // 第二种策略: 分割数字分别识别
        String result2 = recognizeEachDigit(prepared, tesseract);
      
        // 选择更合理的结果
        int confidence1 = calculateConfidence(result1);
        int confidence2 = calculateConfidence(result2);
      
        return confidence1 > confidence2 ? result1 : result2;
    }
  
    // 逐个数字识别
    private static String recognizeEachDigit(Mat image, ITesseract tesseract) {
        // 1. 查找轮廓
        List<MatOfPoint> contours = new ArrayList<>();
        Mat hierarchy = new Mat();
        Imgproc.findContours(image.clone(), contours, hierarchy, Imgproc.RETR_EXTERNAL, Imgproc.CHAIN_APPROX_SIMPLE);
      
        // 2. 轮廓过滤和排序 (从左到右)
        contours.removeIf(cnt -> {
            Rect rect = Imgproc.boundingRect(cnt);
            return rect.width < 10 || rect.height < 20;
        });
      
        Collections.sort(contours, new Comparator<MatOfPoint>() {
            @Override
            public int compare(MatOfPoint c1, MatOfPoint c2) {
                Rect r1 = Imgproc.boundingRect(c1);
                Rect r2 = Imgproc.boundingRect(c2);
                return Integer.compare(r1.x, r2.x);
            }
        });
      
        // 3. 创建结果缓存
        StringBuilder result = new StringBuilder();
      
        // 4. 分别处理每个数字区域
        for (MatOfPoint contour : contours) {
            Rect rect = Imgproc.boundingRect(contour);
          
            // 扩展边界确保完整数字
            Rect expanded = expandRect(rect, image.size(), 3);
            Mat digitRoi = new Mat(image, expanded);
          
            // 识别数字
            String digitResult = performOCR(digitRoi, tesseract);
          
            // 置信度评估
            if (isLikelyDigit(digitResult)) {
                result.append(digitResult.replaceAll("\\D", "")); // 只保留数字
            }
        }
      
        return result.toString();
    }

    /** ============================ 结果处理 ============================ */
  
    private static String extractDelayValue(String text) {
        if (text == null || text.trim().isEmpty()) {
            return "N/A";
        }
      
        // 提取最长的数字序列
        String digits = text.replaceAll("\\D", "");
        String longest = "";
        StringBuilder current = new StringBuilder();
      
        for (char c : digits.toCharArray()) {
            if (Character.isDigit(c)) {
                current.append(c);
                if (current.length() > longest.length()) {
                    longest = current.toString();
                }
            } else {
                current.setLength(0);
            }
        }
      
        // 仅保留1-2位数字
        if (longest.length() > 2) {
            // 取最后两位 - 假设新数值覆盖旧值
            return longest.substring(longest.length() - 2);
        }
      
        return longest.isEmpty() ? "N/A" : longest;
    }

    // 置信度评估
    private static int calculateConfidence(String text) {
        // 1. 文本长度符合预期
        int lengthScore = Math.min(3, text.length()) * 20;
      
        // 2. 数字化程度
        int digitScore = (int) (text.replaceAll("\\D", "").length() * 10.0 / text.length());
      
        // 3. 额外逻辑判断
        int logicScore = text.contains("ms") ? 30 : (text.startsWith("MS") ? 15 : 0);
      
        return lengthScore + digitScore + logicScore;
    }
  
    // 可能是数字的评估
    private static boolean isLikelyDigit(String text) {
        return text.matches(".*\\d.*"); // 包含至少一个数字

    // 矩形扩展
    private static Rect expandRect(Rect rect, Size maxSize, int padding) {
        int x = (int) Math.max(0, rect.x - padding);
        int y = (int) Math.max(0, rect.y - padding);
        int width = (int) Math.min(maxSize.width - x - 1, rect.width + padding * 2);
        int height = (int) Math.min(maxSize.height - y - 1, rect.height + padding * 2);
        return new Rect(x, y, width, height);
    }
```

## 核心优化技术

### 1. 图像分割与轮廓检测

2. **轮廓过滤**：基于面积筛选掉噪点和小区域
3. **轮廓填充**：drawContours带FILLED选项填充数字内部空洞

### 3. 数字分割与独立识别

```java
// 分割并分别识别每个数字
```

2. **边界扩展**：使用expandRect确保完整捕获数字所有部分
3. **独立识别**：对每个数字单独进行OCR识别，提高准确率
