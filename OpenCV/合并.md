# 改进的数字分割与修复方案：智能轮廓合并与校正

为了解决多分割问题和m/s字符不完整的问题，我设计了一个智能轮廓合并系统，根据轮廓大小和相对位置进行自动合并校正。以下是完整的解决方案：

```java
import org.opencv.core.*;
import org.opencv.imgproc.Imgproc;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class ImprovedDigitSegmenter {

    // 分割阈值参数 (可根据实际图像调整)
    private static final double MIN_CONTOUR_AREA = 300;    // 最小轮廓面积
    private static final double MAX_CONTOUR_AREA = 5000;   // 最大轮廓面积
    private static final double MAX_WIDTH_RATIO = 2.5;     // 宽高最大比例
    private static final double MAX_DISTANCE_RATIO = 1.5;  // 相邻轮廓最大距离比例
    private static final int MAX_SEPARATED_PARTS = 3;      // 最多允许分离部分
  
    // 图像放大系数
    private static final int SCALE_FACTOR = 6;

    /**
     * 改进的数字分割方法 - 带轮廓合并机制
     * 
     * @param originalImage 原始图像(已放大)
     * @param originalRect 原始矩形区域(未放大)
     * @return 分割修复后的数字图像列表
     */
    public static List<Mat> segmentAndRepairDigits(Mat originalImage, Rect originalRect) {
        // 1. 计算放大后的尺寸
        int scaledWidth = originalRect.width * SCALE_FACTOR;
        int scaledHeight = originalRect.height * SCALE_FACTOR;
      
        // 2. 预处理: 二值化
        Mat binaryImage = preprocessImage(originalImage);
      
        // 3. 查找所有轮廓
        List<MatOfPoint> contours = findContours(binaryImage.clone());
      
        // 4. 初步筛选并排序轮廓
        List<DigitContour> digitContours = filterAndSortContours(contours, scaledWidth);
      
        // 5. 智能轮廓合并 (核心)
        List<DigitContour> mergedContours = mergeBrokenContours(digitContours, scaledHeight);
      
        // 6. 提取和修复每个数字区域
        return extractDigitImages(binaryImage, mergedContours);
    }

    /**
     * 图像预处理: 灰度化 + 二值化
     */
    private static Mat preprocessImage(Mat image) {
        Mat gray = new Mat();
        Mat binary = new Mat();
      
        // 如果已是灰度图则跳过转换
        if (image.channels() > 1) {
            Imgproc.cvtColor(image, gray, Imgproc.COLOR_BGR2GRAY);
        } else {
            image.copyTo(gray);
        }
      
        // 自适应阈值二值化
        Imgproc.adaptiveThreshold(
            gray, binary, 255,
            Imgproc.ADAPTIVE_THRESH_GAUSSIAN_C, 
            Imgproc.THRESH_BINARY_INV, 
            15, 
            7
        );
      
        return binary;
    }

    /**
     * 查找所有轮廓
     */
    private static List<MatOfPoint> findContours(Mat binaryImage) {
        List<MatOfPoint> contours = new ArrayList<>();
        Mat hierarchy = new Mat();
        Imgproc.findContours(binaryImage, contours, hierarchy, 
                            Imgproc.RETR_EXTERNAL, Imgproc.CHAIN_APPROX_SIMPLE);
        return contours;
    }

    /**
     * 轮廓过滤与排序
     */
    private static List<DigitContour> filterAndSortContours(List<MatOfPoint> contours, int scaledWidth) {
        List<DigitContour> digitContours = new ArrayList<>();
      
        for (MatOfPoint contour : contours) {
            Rect rect = Imgproc.boundingRect(contour);
            double area = rect.area();
          
            // 轮廓面积过滤
            if (area < MIN_CONTOUR_AREA || area > MAX_CONTOUR_AREA) continue;
          
            // 宽高比过滤（避免过宽的非字符区域）
            double aspectRatio = (double)rect.width / rect.height;
            if (aspectRatio > MAX_WIDTH_RATIO) continue;
          
            // 创建轮廓包装对象
            digitContours.add(new DigitContour(contour, rect, area));
        }
      
        // 按水平位置排序 (从左到右)
        Collections.sort(digitContours, Comparator.comparingInt(dc -> dc.rect.x));
      
        return digitContours;
    }

    /**
     * 智能轮廓合并 (核心算法)
     */
    private static List<DigitContour> mergeBrokenContours(List<DigitContour> contours, int imageHeight) {
        if (contours.size() < 2) return contours;
      
        List<DigitContour> mergedContours = new ArrayList<>();
        DigitContour current = contours.get(0);
      
        for (int i = 1; i < contours.size(); i++) {
            DigitContour next = contours.get(i);
          
            // 判断是否应该合并
            if (shouldMergeContours(current, next, imageHeight)) {
                current = mergeTwoContours(current, next);
                // 如果下一个也是需要合并的，继续循环而不要添加
                continue;
            }
          
            // 检查当前轮廓是否由多个小部分组成
            DigitContour checked = checkMultiPartContour(current);
            mergedContours.add(checked);
            current = next;
        }
      
        // 添加最后一个轮廓
        mergedContours.add(checkMultiPartContour(current));
      
        return mergedContours;
    }

    /**
     * 判断两个轮廓是否应该合并
     */
    private static boolean shouldMergeContours(DigitContour first, DigitContour second, int imageHeight) {
        Rect r1 = first.rect;
        Rect r2 = second.rect;
      
        // 计算水平距离和垂直距离
        double horizontalDist = r2.x - (r1.x + r1.width);
        double verticalDist = Math.abs((r1.y + r1.height/2) - (r2.y + r2.height/2));
      
        // 垂直重叠比例
        double verticalOverlap = calculateVerticalOverlap(r1, r2);
      
        // 判断条件1: 水平距离很小且垂直位置相近
        boolean isCloseInX = horizontalDist < (r1.height * MAX_DISTANCE_RATIO);
        boolean isAligned = verticalDist < (imageHeight * 0.25);
        boolean shouldMerge1 = isCloseInX && isAligned;
      
        // 判断条件2: 垂直方向有显著重叠
        boolean significantOverlap = verticalOverlap > 0.6;
        boolean sameAreaRange = Math.abs(r1.area() - r2.area()) < (Math.min(r1.area(), r2.area()) * 0.7);
        boolean shouldMerge2 = significantOverlap && sameAreaRange;
      
        // 合并逻辑: 如果任意条件满足，并且两者面积都不大
        boolean bothSmall = first.area < 1200 && second.area < 1200;
        boolean currentSize = mergedContourSize(first) < MAX_SEPARATED_PARTS;
      
        return (shouldMerge1 || shouldMerge2) && bothSmall && currentSize;
    }

    /**
     * 计算两个矩形的垂直重叠比例
     */
    private static double calculateVerticalOverlap(Rect r1, Rect r2) {
        int y1 = r1.y;
        int y1_end = y1 + r1.height;
        int y2 = r2.y;
        int y2_end = y2 + r2.height;
      
        int maxStart = Math.max(y1, y2);
        int minEnd = Math.min(y1_end, y2_end);
      
        if (minEnd <= maxStart) return 0.0;
      
        double overlap = minEnd - maxStart;
        return overlap / Math.max(r1.height, r2.height);
    }

    /**
     * 合并两个轮廓
     */
    private static DigitContour mergeTwoContours(DigitContour first, DigitContour second) {
        Rect r1 = first.rect;
        Rect r2 = second.rect;
      
        // 计算合并后的矩形
        int x = Math.min(r1.x, r2.x);
        int y = Math.min(r1.y, r2.y);
        int width = Math.max(r1.x + r1.width, r2.x + r2.width) - x;
        int height = Math.max(r1.y + r1.height, r2.y + r2.height) - y;
        Rect mergedRect = new Rect(x, y, width, height);
      
        // 创建新的轮廓容器
        DigitContour merged = new DigitContour(null, mergedRect, first.area + second.area);
        merged.isCombined = true;
      
        // 记录原始轮廓用于调试
        merged.sourceContours.add(first.originalContour);
        merged.sourceContours.add(second.originalContour);
      
        if (first.sourceContours != null) {
            merged.sourceContours.addAll(first.sourceContours);
        }
      
        if (second.sourceContours != null) {
            merged.sourceContours.addAll(second.sourceContours);
        }
      
        return merged;
    }

    /**
     * 检查多部分轮廓
     */
    private static DigitContour checkMultiPartContour(DigitContour contour) {
        if (contour.sourceContours.size() > 1) {
            // 对于已合并的轮廓，进行后处理修复
            contour.requiresRepair = true;
        }
        return contour;
    }

    /**
     * 获取合并轮廓的大小
     */
    private static int mergedContourSize(DigitContour contour) {
        return contour.sourceContours == null ? 1 : contour.sourceContours.size() + 1;
    }

    /**
     * 提取数字图像并修复
     */
    private static List<Mat> extractDigitImages(Mat binaryImage, List<DigitContour> mergedContours) {
        List<Mat> digitImages = new ArrayList<>();
      
        for (DigitContour digitContour : mergedContours) {
            Rect rect = digitContour.rect;
          
            // 扩展矩形区域确保包含整个字符
            expandRect(rect, binaryImage.size(), 3);
          
            Mat digitMat = new Mat(binaryImage, rect);
          
            // 修复断裂部分 (特别是对于合并的轮廓)
            if (digitContour.requiresRepair || digitContour.isCombined) {
                Mat repaired = repairDigit(digitMat);
                digitImages.add(repaired);
            } else {
                // 调整到标准尺寸
                Mat resized = new Mat();
                Imgproc.resize(digitMat, resized, new Size(28, 28));
                digitImages.add(resized);
            }
        }
      
        return digitImages;
    }

    /**
     * 扩展矩形区域
     */
    private static void expandRect(Rect rect, Size maxSize, int padding) {
        rect.x = Math.max(0, rect.x - padding);
        rect.y = Math.max(0, rect.y - padding);
        rect.width = Math.min((int)maxSize.width - rect.x - 1, rect.width + padding * 2);
        rect.height = Math.min((int)maxSize.height - rect.y - 1, rect.height + padding * 2);
    }

    /**
     * 修复数字断裂部分 (改进版)
     */
    private static Mat repairDigit(Mat digitImage) {
        // 1. 根据字符方向选择不同的核
        Size kernelSize;
        if (digitImage.width() * 1.5 < digitImage.height()) {
            // 纵向字符 (可能是"I","1")
            kernelSize = new Size(1, 3);
        } else if (digitImage.width() > digitImage.height() * 1.5) {
            // 横向字符 (可能是"-")
            kernelSize = new Size(6, 1);
        } else {
            // 常规字符
            kernelSize = new Size(3, 3);
        }
      
        // 2. 自适应形态学闭合
        Mat kernel = Imgproc.getStructuringElement(Imgproc.MORPH_ELLIPSE, kernelSize);
      
        // 3. 执行闭运算 (先膨胀后腐蚀)
        for (int i = 0; i < 2; i++) {
            Imgproc.dilate(digitImage, digitImage, kernel);
        }
        for (int i = 0; i < 1; i++) {
            Imgproc.erode(digitImage, digitImage, kernel);
        }
      
        // 4. 中值滤波去噪
        Mat result = new Mat();
        Imgproc.medianBlur(digitImage, result, 3);
      
        // 5. 调整到标准尺寸
        Imgproc.resize(result, result, new Size(28, 28));
      
        return result;
    }

    /**
     * 数字轮廓容器类 (辅助数据结构)
     */
    static class DigitContour {
        MatOfPoint originalContour;  // 原始轮廓
        Rect rect;                   // 轮廓边界矩形
        double area;                 // 轮廓面积
        boolean isCombined = false;      // 是否是合并的轮廓
        boolean requiresRepair = false;  // 是否需要修复
        List<MatOfPoint> sourceContours = new ArrayList<>();  // 原始轮廓列表(用于合并的轮廓)

        DigitContour(MatOfPoint contour, Rect rect, double area) {
            this.originalContour = contour;
            this.rect = rect;
            this.area = area;
            if (contour != null) {
                this.sourceContours.add(contour);
            }
        }
    }
}
```
