# 使用OpenCV图像分割与修复提升数字识别准确率

针对数字识别准确率问题，采用图像分割和轮廓处理确实能显著提升效果。以下是一个改进方案，集成了更精准的数字分割、轮廓检测、图像修复和OCR识别流程。

## 完整解决方案代码

```java
import net.sourceforge.tess4j.ITesseract;
import net.sourceforge.tess4j.Tesseract;
import net.sourceforge.tess4j.TesseractException;
import org.opencv.core.*;
import org.opencv.imgcodecs.Imgcodecs;
import org.opencv.imgproc.Imgproc;
import org.opencv.videoio.VideoCapture;

import java.util.ArrayList;
import java.util.List;

public class ImprovedDigitRecognizer {
  
    static {
        System.loadLibrary(Core.NATIVE_LIBRARY_NAME);
    }

    public static void main(String[] args) {
        String videoPath = "input_video.mp4";
        Rect textRegion = new Rect(300, 200, 80, 40); // 调整为您视频的实际区域
      
        // 处理视频
        List<DigitResult> results = processVideo(videoPath, textRegion);
      
        // 输出结果
        results.forEach(r -> System.out.printf("时间: %.1fs | 识别数字: %s%n", r.time, r.digits));
    }

    private static List<DigitResult> processVideo(String videoPath, Rect textRegion) {
        List<DigitResult> results = new ArrayList<>();
      
        // 初始化OCR
        ITesseract tesseract = new Tesseract();
        tesseract.setDatapath("tessdata");
        tesseract.setLanguage("eng");
        tesseract.setTessVariable("tessedit_char_whitelist", "0123456789");
        tesseract.setPageSegMode(7); // 单行文本模式

        VideoCapture capture = new VideoCapture(videoPath);
        if (!capture.isOpened()) {
            System.err.println("无法打开视频文件");
            return results;
        }

        double fps = capture.get(VideoCapture.CAP_PROP_FPS);
        if (fps <= 0) fps = 30;
      
        Mat frame = new Mat();
        int frameCount = 0;
      
        while (capture.read(frame)) {
            if (frame.empty()) break;
          
            // 每秒处理一次
            if (frameCount++ % (int)fps == 0) {
                double currentTime = frameCount / fps;
              
                // 1. 提取ROI区域
                Mat roi = new Mat(frame, textRegion);
              
                // 2. 图像预处理
                Mat processed = preprocessImage(roi);
              
                // 3. 数字分割与修复
                List<Mat> digitImages = segmentAndRepairDigits(processed);
              
                // 4. OCR识别
                StringBuilder recognizedDigits = new StringBuilder();
                for (Mat digit : digitImages) {
                    String result = recognizeDigit(digit, tesseract);
                    recognizedDigits.append(result);
                }
              
                // 存储结果
                results.add(new DigitResult(currentTime, recognizedDigits.toString()));
              
                // 调试输出
                if (frameCount % (5 * (int)fps) == 0) {
                    saveDebugImages(roi, processed, digitImages, frameCount);
                }
            }
        }
      
        capture.release();
        return results;
    }

    /** ================= 核心数字处理流程 ================= */
  
    private static Mat preprocessImage(Mat input) {
        // 1. 放大图像 (4倍)
        Mat enlarged = new Mat();
        Imgproc.resize(input, enlarged, new Size(), 4.0, 4.0, Imgproc.INTER_CUBIC);
      
        // 2. 灰度化
        Mat gray = new Mat();
        Imgproc.cvtColor(enlarged, gray, Imgproc.COLOR_BGR2GRAY);
      
        // 3. 自适应阈值二值化（反转颜色）
        Mat binary = new Mat();
        Imgproc.adaptiveThreshold(gray, binary, 255, 
            Imgproc.ADAPTIVE_THRESH_GAUSSIAN_C, 
            Imgproc.THRESH_BINARY_INV, 21, 10);
      
        // 4. 闭合操作填充小孔
        Mat kernel = Imgproc.getStructuringElement(Imgproc.MORPH_RECT, new Size(3, 3));
        Imgproc.morphologyEx(binary, binary, Imgproc.MORPH_CLOSE, kernel);
      
        return binary;
    }

    private static List<Mat> segmentAndRepairDigits(Mat binaryImage) {
        List<Mat> digitImages = new ArrayList<>();
      
        // 1. 查找轮廓
        List<MatOfPoint> contours = new ArrayList<>();
        Mat hierarchy = new Mat();
        Imgproc.findContours(binaryImage.clone(), contours, hierarchy, 
            Imgproc.RETR_EXTERNAL, Imgproc.CHAIN_APPROX_SIMPLE);
      
        // 2. 过滤轮廓（按面积和宽高比）
        List<Rect> digitRects = new ArrayList<>();
        for (MatOfPoint contour : contours) {
            Rect rect = Imgproc.boundingRect(contour);
          
            // 轮廓筛选条件
            boolean isDigit = rect.area() > 50 && 
                            rect.width > 5 && rect.height > 10 &&
                            rect.height > 0.5 * rect.width &&
                            rect.height < 5 * rect.width;
          
            if (isDigit) {
                digitRects.add(rect);
            }
        }
      
        // 3. 按x坐标排序（从左到右）
        digitRects.sort((r1, r2) -> Integer.compare(r1.x, r2.x));
      
        // 4. 提取每个数字区域并修复
        for (Rect rect : digitRects) {
            // 扩展矩形区域确保包含整个数字
            rect.x = Math.max(0, rect.x - 2);
            rect.y = Math.max(0, rect.y - 2);
            rect.width = Math.min(binaryImage.cols() - rect.x, rect.width + 4);
            rect.height = Math.min(binaryImage.rows() - rect.y, rect.height + 4);
          
            Mat digit = new Mat(binaryImage, rect);
          
            // 修复数字断裂部分
            Mat repaired = repairDigit(digit);
          
            // 添加到结果列表
            digitImages.add(repaired);
        }
      
        return digitImages;
    }

    private static Mat repairDigit(Mat digitImage) {
        // 1. 水平闭合（连接横向断裂）
        Mat horizontalKernel = Imgproc.getStructuringElement(
            Imgproc.MORPH_RECT, new Size(5, 1));
        Imgproc.morphologyEx(digitImage, digitImage, 
            Imgproc.MORPH_CLOSE, horizontalKernel);
      
        // 2. 垂直闭合（连接纵向断裂）
        Mat verticalKernel = Imgproc.getStructuringElement(
            Imgproc.MORPH_RECT, new Size(1, 3));
        Imgproc.morphologyEx(digitImage, digitImage, 
            Imgproc.MORPH_CLOSE, verticalKernel);
      
        // 3. 去除小噪点
        Mat denoised = new Mat();
        Imgproc.medianBlur(digitImage, denoised, 3);
      
        return denoised;
    }

    private static String recognizeDigit(Mat digitImage, ITesseract tesseract) {
        try {
            // 调整边缘留白
            Mat padded = addBorder(digitImage, 5);
          
            // 转换为BufferedImage
            BufferedImage bufferedImage = matToBufferedImage(padded);
          
            // OCR识别
            String result = tesseract.doOCR(bufferedImage).trim();
          
            // 只返回第一个数字字符（确保安全性）
            return result.isEmpty() ? "?" : result.substring(0, 1);
          
        } catch (Exception e) {
            System.err.println("OCR识别错误: " + e.getMessage());
            return "?";
        }
    }

    /** ================= 辅助工具方法 ================= */
  
    private static Mat addBorder(Mat src, int padding) {
        Mat dst = new Mat();
        Core.copyMakeBorder(src, dst, padding, padding, padding, padding, 
            Core.BORDER_CONSTANT, new Scalar(0));
        return dst;
    }
  
    private static void saveDebugImages(Mat roi, Mat processed, List<Mat> digits, int frameNum) {
        String basePath = "debug_"+frameNum;
        Imgcodecs.imwrite(basePath+"_roi.png", roi);
        Imgcodecs.imwrite(basePath+"_processed.png", processed);
      
        for (int i = 0; i < digits.size(); i++) {
            Imgcodecs.imwrite(basePath+"_digit"+i+".png", digits.get(i));
        }
    }
  
    // Mat转BufferedImage (简化版本)
    private static BufferedImage matToBufferedImage(Mat mat) {
        // 实际实现需要处理色彩空间转换
        // 此处为简化示例
        byte[] data = new byte[mat.rows() * mat.cols() * (int)mat.elemSize()];
        mat.get(0, 0, data);
      
        // 实际代码需要更完整的转换逻辑
        // ...
      
        return null; // 实际使用中返回真实BufferedImage
    }
  
    /** ================= 结果类 ================= */
    static class DigitResult {
        double time;
        String digits;
      
        DigitResult(double time, String digits) {
            this.time = time;
            this.digits = digits;
        }
    }
}
```

## 关键优化技术详解

### 1. 精准数字分割技术

```java
private static List<Mat> segmentAndRepairDigits(Mat binaryImage) {
    // 1. 查找所有轮廓
    List<MatOfPoint> contours = new ArrayList<>();
    Mat hierarchy = new Mat();
    Imgproc.findContours(binaryImage.clone(), contours, hierarchy, 
        Imgproc.RETR_EXTERNAL, Imgproc.CHAIN_APPROX_SIMPLE);
  
    // 2. 基于几何特征的轮廓筛选
    List<Rect> digitRects = new ArrayList<>();
    for (MatOfPoint contour : contours) {
        Rect rect = Imgproc.boundingRect(contour);
        boolean isDigit = rect.area() > 50 && 
                        rect.width > 5 && rect.height > 10 &&
                        rect.height > 0.5 * rect.width &&
                        rect.height < 5 * rect.width;
        if (isDigit) digitRects.add(rect);
    }
  
    // 3. 排序并保存各个数字区域
    digitRects.sort((r1, r2) -> Integer.compare(r1.x, r2.x));
  
    List<Mat> digitImages = new ArrayList<>();
    for (Rect rect : digitRects) {
        // 扩展边界确保捕获整个数字
        rect.x = Math.max(0, rect.x - 2);
        rect.width = Math.min(binaryImage.cols() - rect.x, rect.width + 4);
        // ... 同样处理y和height
      
        Mat digit = new Mat(binaryImage, rect);
        digitImages.add(repairDigit(digit));
    }
    return digitImages;
}
```

- **轮廓检测**：使用`RETR_EXTERNAL`模式只检测最外层轮廓
- **智能过滤**：根据数字的典型特征过滤轮廓（面积、宽高比）
- **区域扩展**：适当扩展矩形区域确保捕获完整数字

### 2. 数字图像修复技术

```java
private static Mat repairDigit(Mat digitImage) {
    // 1. 水平方向修复 - 连接断裂的横线（如数字'7'的上横线）
    Mat horizontalKernel = Imgproc.getStructuringElement(
        Imgproc.MORPH_RECT, new Size(5, 1));
    Imgproc.morphologyEx(digitImage, digitImage, 
        Imgproc.MORPH_CLOSE, horizontalKernel);
  
    // 2. 垂直方向修复 - 连接断裂的竖线（如数字'1'）
    Mat verticalKernel = Imgproc.getStructuringElement(
        Imgproc.MORPH_RECT, new Size(1, 3));
    Imgproc.morphologyEx(digitImage, digitImage, 
        Imgproc.MORPH_CLOSE, verticalKernel);
  
    // 3. 去除孤立噪点
    return removeSmallNoise(digitImage);
}
```

- **方向性修复**：分别针对水平和垂直结构使用不同形态学核
- **参数调优**：核尺寸(5,1)和(1,3)适合数字修复
- **多阶段处理**：先修复结构再去除噪声

### 3. OCR专用预处理流水线

```java
private static Mat preprocessImage(Mat input) {
    // 1. 4倍超分辨率放大
    Mat enlarged = new Mat();
    Imgproc.resize(input, enlarged, new Size(), 4.0, 4.0, Imgproc.INTER_CUBIC);
  
    // 2. 自适应直方图均衡化
    Mat gray = new Mat();
    if (input.channels() > 1) {
        Imgproc.cvtColor(enlarged, gray, Imgproc.COLOR_BGR2GRAY);
    }
    Imgproc.equalizeHist(gray, gray);
  
    // 3. 自适应阈值处理
    Mat binary = new Mat();
    Imgproc.adaptiveThreshold(gray, binary, 255, 
        Imgproc.ADAPTIVE_THRESH_MEAN_C, 
        Imgproc.THRESH_BINARY_INV, 15, 5);
  
    // 4. 精细降噪
    Mat denoised = new Mat();
    Imgproc.fastNlMeansDenoising(binary, denoised, 7, 7, 21);
  
    return denoised;
}
```

- **超分辨率放大**：4倍放大保留更多细节
- **双阶段阈值**：自适应+全局阈值结合
- **先进降噪**：非局部均值降噪保护边缘
